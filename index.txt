        let audioCtx = null;
        function getAudioContext() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
            if (audioCtx.state === 'suspended') {
                audioCtx.resume();
            }
            return audioCtx;
        }

        function playSound(type) {
            if (state.soundMuted) return;
            try {
                const ctx = getAudioContext();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.connect(gain);
                gain.connect(ctx.destination);
                const now = ctx.currentTime;

                if (type === 'shift') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(440, now);
                    osc.frequency.exponentialRampToValueAtTime(660, now + 0.08);
                    gain.gain.setValueAtTime(0.12, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.08);
                    osc.start(now);
                    osc.stop(now + 0.08);
                } else if (type === 'complete') {
                    osc.type = 'triangle';
                    osc.frequency.setValueAtTime(523.25, now);
                    osc.frequency.setValueAtTime(659.25, now + 0.08);
                    osc.frequency.setValueAtTime(783.99, now + 0.16);
                    gain.gain.setValueAtTime(0.18, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.3);
                    osc.start(now);
                    osc.stop(now + 0.3);
                } else if (type === 'click') {
                    osc.type = 'sine';
                    osc.frequency.setValueAtTime(320, now);
                    gain.gain.setValueAtTime(0.06, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.04);
                    osc.start(now);
                    osc.stop(now + 0.04);
                } else if (type === 'delete') {
                    osc.type = 'sawtooth';
                    osc.frequency.setValueAtTime(220, now);
                    osc.frequency.exponentialRampToValueAtTime(110, now + 0.15);
                    gain.gain.setValueAtTime(0.12, now);
                    gain.gain.linearRampToValueAtTime(0.01, now + 0.15);
                    osc.start(now);
                    osc.stop(now + 0.15);
                }
            } catch(e) { console.error(e); }
        }

        function enableBarDragging(barElement, task) {
            let startX = 0;
            let initialStartDay = task.startDay;
            const columnWidths = { compact: 36, standard: 54, detailed: 72 };
            const colWidth = columnWidths[state.zoomScale] || 54;

            barElement.onmousedown = (e) => {
                e.preventDefault();
                startX = e.clientX;
                initialStartDay = task.startDay;

                const onMouseMove = (moveEvent) => {
                    const deltaX = moveEvent.clientX - startX;
                    const dayDelta = Math.round(deltaX / colWidth);
                    const newStartDay = Math.max(1, Math.min(34 - task.duration, initialStartDay + dayDelta));

                    if (newStartDay !== task.startDay) {
                        task.startDay = newStartDay;
                        playSound('shift');
                        renderGanttChart();
                    }
                };

                const onMouseUp = () => {
                    document.removeEventListener('mousemove', onMouseMove);
                    document.removeEventListener('mouseup', onMouseUp);
                    saveToStorage();
                };

                document.addEventListener('mousemove', onMouseMove);
                document.addEventListener('mouseup', onMouseUp);
            };
        }

        function shiftTask(taskId, delta) {
            const proj = getCurrentProject();
            const task = proj.tasks.find(t => t.id === taskId);
            if (task) {
                task.startDay = Math.max(1, Math.min(34 - task.duration, task.startDay + delta));
                playSound('shift');
                saveToStorage();
                renderGanttChart();
            }
        }

            const proj = getCurrentProject();

        function quickUpdateProgress(taskId) {
            const proj = getCurrentProject();
            const task = proj.tasks.find(t => t.id === taskId);
            if (task) {
                let newProg = prompt(`Update progress % for "${task.title}":`, task.progress);
                if (newProg !== null) {
                    newProg = Math.max(0, Math.min(100, parseInt(newProg) || 0));
                    task.progress = newProg;
                    if (newProg === 100) {
                        playSound('complete');
                        triggerConfetti();
                    } else {
                        playSound('click');

                    saveToStorage();
                    renderGanttChart();

            }
        }

            const assignee = document.getElementById('modal-task-assignee').value;
            const points = parseInt(document.getElementById('modal-task-points').value) || 1;
            const startDay = parseInt(document.getElementById('modal-task-start').value) || 1;
            const duration = parseInt(document.getElementById('modal-task-duration').value) || 1;
            const priority = document.getElementById('modal-task-priority').value;
            const progress = parseInt(document.getElementById('modal-task-progress').value) || 0;
            const isMilestone = document.getElementById('modal-task-milestone').value === "true";

            if (taskId) {
                const task = proj.tasks.find(t => t.id === taskId);
                if (task) {
                    task.title = title;
                    task.assignee = assignee;
                    task.points = points;
                    task.startDay = startDay;
                    task.duration = duration;
                    task.priority = priority;
                    task.progress = progress;
                    task.isMilestone = isMilestone;
                }
            } else {
                const newTask = {
                    id: 't-' + Date.now(),
                    title, assignee, points, startDay, duration, priority, progress, isMilestone,
                    revisionDays: 0, subtasks: []
                };
                proj.tasks.push(newTask);
            }

            if (progress === 100) triggerConfetti();
            saveToStorage();
            closeModal('taskModal');
            renderGanttChart();
        }

        function renderKanbanBoard() {
            const container = document.getElementById('kanban-board-container');
            container.innerHTML = '';
            const proj = getCurrentProject();

            const cols = [
                { title: 'To Do', filter: t => t.progress === 0 },
                { title: 'In Progress', filter: t => t.progress > 0 && t.progress < 100 },
                { title: 'In Review', filter: t => t.revisionDays > 0 },
                { title: 'Completed', filter: t => t.progress === 100 }
            ];

            cols.forEach(col => {
                const tasksInCol = proj.tasks.filter(col.filter);
                const colHtml = `
                    <div class="bg-slate-100 dark:bg-slate-800/60 p-3 rounded-2xl border border-slate-200/80 dark:border-slate-800 flex flex-col gap-3 min-h-[280px]">
                        <div class="flex items-center justify-between font-bold text-xs text-slate-700 dark:text-slate-300 pb-2 border-b border-slate-200 dark:border-slate-700">
                            <span>${col.title}</span>
                            <span class="px-2 py-0.5 rounded-full bg-slate-200 dark:bg-slate-700 text-[10px]">${tasksInCol.length}</span>
                        </div>
                        <div class="flex flex-col gap-2 flex-1">
                            ${tasksInCol.map(t => `
                                <div class="bg-white dark:bg-darkcard p-3 rounded-xl border border-slate-200/80 dark:border-slate-800 shadow-xs flex flex-col gap-2 text-xs">
                                    <div class="font-semibold text-slate-900 dark:text-white">${t.title}</div>
                                    <div class="flex items-center justify-between text-[10px] text-slate-500">
                                        <span>${t.assignee}</span>
                                        <span class="font-bold text-blue-500">${t.progress}%</span>
                                    </div>
                                </div>
                            `).join('')}
                        </div>
                    </div>
                `;
                container.insertAdjacentHTML('beforeend', colHtml);
            });
        }

        function renderMindMap() {
            const svg = document.getElementById('mindmap-svg');
            svg.innerHTML = '';
            const proj = getCurrentProject();

            const width = svg.clientWidth || 800;
            const height = svg.clientHeight || 400;
            const centerX = width / 2;
            const centerY = height / 2;

            let svgContent = `
                <g>
                    <rect x="${centerX - 90}" y="${centerY - 25}" width="180" height="50" rx="15" fill="#3b82f6" />
                    <text x="${centerX}" y="${centerY + 5}" fill="#ffffff" font-size="12" font-weight="bold" text-anchor="middle">${proj.title.substring(0, 22)}</text>
                </g>
            `;

            const tasks = proj.tasks.slice(0, 8);
            const radius = 170;
            tasks.forEach((t, i) => {
                const angle = (i / tasks.length) * (2 * Math.PI);
                const x = centerX + radius * Math.cos(angle);
                const y = centerY + radius * Math.sin(angle);

                svgContent += `<line x1="${centerX}" y1="${centerY}" x2="${x}" y2="${y}" stroke="${t.progress === 100 ? '#10b981' : '#64748b'}" stroke-width="2" stroke-dasharray="4" />`;
                svgContent += `
                    <g>
                        <rect x="${x - 65}" y="${y - 18}" width="130" height="36" rx="10" fill="${t.progress === 100 ? '#10b981' : '#1e293b'}" stroke="#334155" />
                        <text x="${x}" y="${y + 4}" fill="#ffffff" font-size="10" font-weight="600" text-anchor="middle">${t.title.substring(0, 15)}</text>
                    </g>
                `;
            });

            svg.innerHTML = svgContent;
        }

        function importJSON(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const imported = JSON.parse(e.target.result);
                    if (Array.isArray(imported)) {
                        state.projects = imported;
                        saveToStorage();
                        renderProjectsBentoGrid();
                        renderGanttChart();
                        alert("Projects data imported successfully!");
                    }
                } catch(err) {
                    alert("Invalid JSON format!");
                }
            };
            reader.readAsText(file);
        }
